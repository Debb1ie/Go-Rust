Rust — Axum
// Cargo.toml: axum = "0.7"
use axum::{routing::get, Router};
 
async fn hello() -> &'static str {
  "⚡ Hello from Rust!"
}
 
#[tokio::main]
async fn main() {
  let app = Router::new()
    .route("/", get(hello));
  let listener =
    tokio::net::TcpListener::bind(
      "0.0.0.0:8080").await.unwrap();
  axum::serve(listener, app)
    .await.unwrap();
}
Go — net/http
// No dependencies needed
package main
 
import (
  "fmt"
  "net/http"
)
 
func hello(w http.ResponseWriter,
  r *http.Request) {
  fmt.Fprintln(w,
    "🚀 Hello from Go!")
}
 
func main() {
  http.HandleFunc("/", hello)
  http.ListenAndServe(
    ":8080", nil)
}
C — Mongoose
// Embedded C web server
#include "mongoose.h"
 
void fn(struct mg_connection *c,
  int ev, void *ev_data) {
  if (ev == MG_EV_HTTP_MSG) {
    mg_http_reply(c, 200,
      "",
      "🔧 Hello from C!");
  }
}
 
int main() {
  struct mg_mgr mgr;
  mg_mgr_init(&mgr);
  mg_http_listen(&mgr,
    "http://0.0.0.0:8080",
    fn, NULL);
  for (;;)
    mg_mgr_poll(&mgr, 1000);
}
